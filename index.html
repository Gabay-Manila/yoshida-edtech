    <!-- JavaScript アプリ制御 & Web Audio API -->

    <script>

        // Web Audio API サウンド発振機能 (437Hz)

        let audioCtx = null;

        let isSoundEnabled = true;

        /**

         * 437Hz のサイン波を再生する関数

         * @param {number} freq - 周波数 (デフォルト 437Hz)

         * @param {number} duration - 再生時間（秒）

         */

        function playTone(freq = 437, duration = 0.12) {

            if (!isSoundEnabled) return;

            try {

                // AudioContext の初期化（ユーザー操作後に利用可能）

                if (!audioCtx) {

                    const AudioContext = window.AudioContext || window.webkitAudioContext;

                    audioCtx = new AudioContext();

                }

                if (audioCtx.state === 'suspended') {

                    audioCtx.resume();

                }

                const osc = audioCtx.createOscillator();

                const gain = audioCtx.createGain();

                osc.type = 'sine';

                osc.frequency.setValueAtTime(freq, audioCtx.currentTime);

                // アタック & リリース エンベロープでクリックノイズ防止

                gain.gain.setValueAtTime(0.01, audioCtx.currentTime);

                gain.gain.exponentialRampToValueAtTime(0.12, audioCtx.currentTime + 0.02);

                gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);

                osc.connect(gain);

                gain.connect(audioCtx.destination);

                osc.start();

                osc.stop(audioCtx.currentTime + duration);

            } catch (e) {

                console.warn('Web Audio API is not supported or blocked:', e);

            }

        }

        /**

         * サウンドのON/OFF切り替え

         */

        function toggleSound() {

            isSoundEnabled = !isSoundEnabled;

            const icon = document.getElementById('soundIcon');

            const label = document.getElementById('soundLabel');

            if (isSoundEnabled) {

                icon.className = 'fa-solid fa-volume-high text-[var(--brass-deco)]';

                label.textContent = '437Hz 音声: ON';

                playTone(437, 0.1);

            } else {

                icon.className = 'fa-solid fa-volume-xmark text-stone-400';

                label.textContent = '437Hz 音声: OFF';

            }

        }

        // フィルタリング機能

        let currentCategory = 'all';

        /**

         * カテゴリによる絞り込み

         */

        function filterCategory(cat) {

            currentCategory = cat;

            // タブの見た目変更

            const tabs = document.querySelectorAll('#categoryTabs button');

            tabs.forEach(tab => {

                if (tab.dataset.target === cat) {

                    tab.classList.add('active');

                } else {

                    tab.classList.remove('active');

                }

            });

            applyFilters();

        }

        /**

         * 検索キーワードによる絞り込み

         */

        function handleSearch() {

            applyFilters();

        }

        /**

         * 統合フィルター処理（カテゴリ + 検索語句）

         */

        function applyFilters() {

            const query = document.getElementById('searchInput').value.toLowerCase().trim();

            const cards = document.querySelectorAll('#appGrid .app-card');

            let visibleCount = 0;

            cards.forEach(card => {

                const categoryMatch = (currentCategory === 'all' || card.dataset.cat === currentCategory);

                const keywords = card.dataset.keywords ? card.dataset.keywords.toLowerCase() : '';

                const title = card.querySelector('h3').textContent.toLowerCase();

                const desc = card.querySelector('p').textContent.toLowerCase();

                const searchMatch = !query || title.includes(query) || desc.includes(query) || keywords.includes(query);

                if (categoryMatch && searchMatch) {

                    card.style.display = 'flex';

                    visibleCount++;

                } else {

                    card.style.display = 'none';

                }

            });

            // 結果ゼロ時の表示切り替え

            const noResultsEl = document.getElementById('noResults');

            if (visibleCount === 0) {

                noResultsEl.classList.remove('hidden');

            } else {

                noResultsEl.classList.add('hidden');

            }

        }

        /**

         * フィルターのリセット

         */

        function resetFilters() {

            document.getElementById('searchInput').value = '';

            filterCategory('all');

            playTone(437, 0.1);

        }
